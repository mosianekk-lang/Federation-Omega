sage.get("role") or "observer"),
                    content=str(content or ""),
                    occurred_at=str(occurred),
                    privacy_tier=str(message.get("privacy_tier") or "P2_CONFIDENTIAL"),
                    case_wall=str(message.get("case_wall") or "GENERAL"),
                    metadata={"import_file": source_path.name},
                )
            )
        return self.ingest_many(events)

    def process_outbox(self, outbox: str | Path) -> dict[str, Any]:
        outbox_path = Path(outbox).expanduser().resolve()
        processed = outbox_path / "processed"
        rejected = outbox_path / "dead_letter"
        processed.mkdir(parents=True, exist_ok=True)
        rejected.mkdir(parents=True, exist_ok=True)
        accepted_count = 0
        rejected_count = 0
        for path in sorted(p for p in outbox_path.iterdir() if p.is_file() and p.suffix in {".json", ".jsonl"}):
            try:
                if path.suffix == ".jsonl":
                    payloads = [json.loads(line) for line in path.read_text(encoding="utf-8").splitlines() if line.strip()]
                else:
                    value = json.loads(path.read_text(encoding="utf-8"))
                    payloads = value if isinstance(value, list) else [value]
                self.ingest_many(payloads)
                destination = processed / path.name
                if destination.exists():
                    destination = processed / f"{path.stem}-{sha256_text(path.read_text(encoding='utf-8'))[:12]}{path.suffix}"
                shutil.move(str(path), str(destination))
                accepted_count += len(payloads)
            except Exception as exc:  # fail closed and preserve payload hash
                try:
                    payload = path.read_text(encoding="utf-8")
                except OSError:
                    payload = "UNREADABLE"
                self.store.record_dead_letter(str(path), payload, str(exc))
                destination = rejected / path.name
                if destination.exists():
                    destination = rejected / f"{path.stem}-{sha256_text(payload)[:12]}{path.suffix}"
                shutil.move(str(path), str(destination))
                rejected_count += 1
        return {"accepted_events": accepted_count, "rejected_files": rejected_count}


class _Handler(BaseHTTPRequestHandler):
    engine: LiveBibleEngine
    token: str

    def _send(self, status: int, value: dict[str, Any]) -> None:
        body = json.dumps(value, ensure_ascii=False).encode("utf-8")
        self.send_response(status)
        origin = self.headers.get("Origin", "")
        if origin.startswith("chrome-extension://") or origin in {"http://127.0.0.1", "http://localhost"}:
            self.send_header("Access-Control-Allow-Origin", origin)
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.send_header("Cache-Control", "no-store")
        self.end_headers()
        self.wfile.write(body)

    def do_OPTIONS(self) -> None:  # noqa: N802
        self.send_response(HTTPStatus.NO_CONTENT)
        origin = self.headers.get("Origin", "")
        if origin.startswith("chrome-extension://"):
            self.send_header("Access-Control-Allow-Origin", origin)
        self.send_header("Access-Control-Allow-Headers", "Content-Type,X-Live-Bible-Token")
        self.send_header("Access-Control-Allow-Methods", "GET,POST,OPTIONS")
        self.end_headers()

    def do_GET(self) -> None:  # noqa: N802
        if urlparse(self.path).path == "/health":
            self._send(HTTPStatus.OK, {"status": "ok", "database_quick_check": self.engine.store.quick_check()})
        else:
            self._send(HTTPStatus.NOT_FOUND, {"error": "not found"})

    def do_POST(self) -> None:  # noqa: N802
        if urlparse(self.path).path != "/v1/events":
            self._send(HTTPStatus.NOT_FOUND, {"error": "not found"})
            return
        supplied = self.headers.get("X-Live-Bible-Token", "")
        if not hmac.compare_digest(supplied, self.token):
            self._send(HTTPStatus.UNAUTHORIZED, {"error": "invalid pairing token"})
            return
        try:
            length = int(self.headers.get("Content-Length", "0"))
            if length <= 0 or length > 5_000_000:
                raise CaptureError("invalid request size")
            value = json.loads(self.rfile.read(length).decode("utf-8"))
            payloads = value if isinstance(value, list) else [value]
            receipts = self.engine.ingest_many(payloads)
            self._send(HTTPStatus.OK, {"receipts": receipts, "count": len(receipts)})
        except (CaptureError, json.JSONDecodeError, UnicodeDecodeError, KeyError, TypeError) as exc:
            self._send(HTTPStatus.BAD_REQUEST, {"error": str(exc)})

    def log_message(self, fmt: str, *args: Any) -> None:
        return


def _token_file(state_dir: Path) -> Path:
    return state_dir / "pairing.token"


def ensure_token(state_dir: Path) -> str:
    state_dir.mkdir(parents=True, exist_ok=True)
    path = _token_file(state_dir)
    if path.exists():
        token = path.read_text(encoding="utf-8").strip()
        if len(token) >= 32:
            return token
    token = secrets.token_urlsafe(32)
    path.write_text(token + "\n", encoding="utf-8")
    try:
        os.chmod(path, 0o600)
    except OSError:
        pass
    return token


def serve(state_dir: Path, host: str, port: int) -> None:
    if host not in {"127.0.0.1", "localhost", "::1"}:
        raise CaptureError("receiver is loopback-only; remote binding is prohibited")
    token = ensure_token(state_dir)
    store = EventStore(state_dir / "live_bible.sqlite3")
    engine = LiveBibleEngine(store, state_dir / "projections")
    handler = type("LiveBibleHandler", (_Handler,), {"engine": engine, "token": token})
    server = ThreadingHTTPServer((host, port), handler)
    print(f"Live Bible receiver listening on http://{host}:{port}")
    print(f"Pairing token path: {_token_file(state_dir)}")
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        pass
    finally:
        server.server_close()


def _state_dir(value: str | None) -> Path:
    return Path(value or os.getenv("LIVE_BIBLE_STATE_DIR") or "~/.live_bible_capture").expanduser().resolve()

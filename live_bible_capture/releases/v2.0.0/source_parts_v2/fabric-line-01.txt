            "schema": event.schema,
            "event_id": event.deterministic_id,
            "source_id": event.source_id,
            "source_type": event.source_type,
            "conversation_id": event.conversation_id,
            "message_id": event.message_id,
            "sequence": event.sequence,
            "role": event.role,
            "content": event.content,
            "content_sha256": sha256_text(event.content),
            "occurred_at": event.occurred_at,
            "observed_at": event.observed_at,
            "privacy_tier": event.privacy_tier,
            "authority_class": event.authority_class,
            "case_wall": event.case_wall,
            "metadata": event.metadata,
        }

    @classmethod
    def from_dict(cls, value: dict[str, Any]) -> "CaptureEvent":
        if not isinstance(value, dict):
            raise CaptureError("capture event must be an object")
        accepted = {
            "schema",
            "source_id",
            "source_type",
            "conversation_id",
            "message_id",
            "sequence",
            "role",
            "content",
            "occurred_at",
            "observed_at",
            "privacy_tier",
            "authority_class",
            "case_wall",
            "metadata",
        }
        unknown = set(value) - accepted - {"event_id", "content_sha256", "chain_hash", "previous_chain_hash"}
        if unknown:
            raise CaptureError(f"unknown capture-event fields: {sorted(unknown)}")
        return cls(
            schema=value.get("schema", SCHEMA),
            source_id=value["source_id"],
            source_type=value.get("source_type", "TURN_CONNECTOR"),
            conversation_id=value["conversation_id"],
            message_id=value["message_id"],
            sequence=value.get("sequence"),
            role=value["role"],
            content=value["content"],
            occurred_at=value["occurred_at"],
            observed_at=value.get("observed_at", utcnow()),
            privacy_tier=value.get("privacy_tier", "P2_CONFIDENTIAL"),
            authority_class=value.get("authority_class", "A1"),
            case_wall=value.get("case_wall", "GENERAL"),
            metadata=value.get("metadata") or {},
        ).normalized()


class EventStore:
    def __init__(self, path: str | Path):
        self.path = Path(path).expanduser().resolve()
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._init_db()
        try:
            os.chmod(self.path, 0o600)
        except OSError:
            pass

    def connect(self) -> sqlite3.Connection:
        connection = sqlite3.connect(self.path, timeout=30, isolation_level=None)
        connection.row_factory = sqlite3.Row
        connection.execute("PRAGMA foreign_keys=ON")
        connection.execute("PRAGMA journal_mode=WAL")
        connection.execute("PRAGMA synchronous=FULL")
        return connection

    def _init_db(self) -> None:
        with self.connect() as db:
            db.executescript(
                """
                CREATE TABLE IF NOT EXISTS events (
                    event_id TEXT PRIMARY KEY,
                    source_id TEXT NOT NULL,
                    source_type TEXT NOT NULL,
                    conversation_id TEXT NOT NULL,
                    message_id TEXT NOT NULL,
                    sequence INTEGER NOT NULL,
                    role TEXT NOT NULL,
                    content TEXT NOT NULL,
                    content_sha256 TEXT NOT NULL,
                    occurred_at TEXT NOT NULL,
                    observed_at TEXT NOT NULL,
                    privacy_tier TEXT NOT NULL,
                    authority_class TEXT NOT NULL,
                    case_wall TEXT NOT NULL,
                    metadata_json TEXT NOT NULL,
                    previous_chain_hash TEXT NOT NULL,
                    chain_hash TEXT NOT NULL UNIQUE,
                    out_of_order INTEGER NOT NULL DEFAULT 0,
                    quarantined INTEGER NOT NULL DEFAULT 0,
                    inserted_at TEXT NOT NULL,
                    UNIQUE(source_id, conversation_id, message_id, content_sha256)
                );
                CREATE INDEX IF NOT EXISTS idx_events_conversation_sequence
                  ON events(conversation_id, sequence, occurred_at, event_id);
                CREATE TABLE IF NOT EXISTS conversation_heads (
                    conversation_id TEXT PRIMARY KEY,
                    head_chain_hash TEXT NOT NULL,
                    max_sequence INTEGER NOT NULL,
                    event_count INTEGER NOT NULL,
                    updated_at TEXT NOT NULL
                );
                CREATE TABLE IF NOT EXISTS gaps (
                    conversation_id TEXT NOT NULL,
                    missing_sequence INTEGER NOT NULL,
                    detected_at TEXT NOT NULL,
                    resolved_at TEXT,
                    PRIMARY KEY(conversation_id, missing_sequence)
                );
                CREATE TABLE IF NOT EXISTS receipts (
                    receipt_id TEXT PRIMARY KEY,
                    event_id TEXT NOT NULL,
                    state TEXT NOT NULL,
                    receipt_json TEXT NOT NULL,
                    created_at TEXT NOT NULL,
                    FOREIGN KEY(event_id) REFERENCES events(event_id)
                );
                CREATE TABLE IF NOT EXISTS dead_letters (
                    dead_letter_id TEXT PRIMARY KEY,
                    source_ref TEXT NOT NULL,
                    error TEXT NOT NULL,
                    payload_sha256 TEXT NOT NULL,
                    created_at TEXT NOT NULL
                );
                """
            )

    def quick_check(self) -> str:
        with self.connect() as db:
            return str(db.execute("PRAGMA quick_check").fetchone()[0])

                receipt["idempotent_replay"] = True
                return receipt

                """,

                    value["source_id"],
                    value["conversation_id"],
                    value["message_id"],
                    value["content_sha256"],
                ),
            ).fetchone()
            if equivalent:
                db.execute("COMMIT")
                receipt = json.loads(equivalent[0])
                receipt["idempotent_replay"] = True
                receipt["equivalent_source_event"] = True
                return receipt
